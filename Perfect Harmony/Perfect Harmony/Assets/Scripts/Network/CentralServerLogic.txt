SAMPLE_PIANO = "piano.wav"
device = "cuda" if torch.cuda.is_available() else "cpu"
print(f"--- [시스템 안내] 감지된 장치: {device.upper()} ---")

def get_normalized_fp(y, sr):
    mfcc = librosa.feature.mfcc(y=y, sr=sr, n_fft=2048, n_mfcc=40)
    mfcc_mean = np.mean(mfcc, axis=1)
    return (mfcc_mean - np.mean(mfcc_mean)) / (np.std(mfcc_mean) + 1e-9)

def run_demucs_separation(audio_bytes):
    with open("temp_input.wav", "wb") as f:
        f.write(audio_bytes)
    print("-> Demucs AI 음원 분리 가동 중 (CPU 안정 모드)...")
    demucs_main(["-n", "htdemucs", "-d", "cpu", "temp_input.wav", "-o", "separated_out"])
    
    base_path = "separated_out/htdemucs/temp_input/"
    y_drums, sr = librosa.load(f"{base_path}drums.wav", sr=None)
    y_bass, _ = librosa.load(f"{base_path}bass.wav", sr=sr)
    y_other, _ = librosa.load(f"{base_path}other.wav", sr=sr)
    return y_drums, y_bass, y_other, sr

@app.post("/generate-chart")
async def generate_chart(file: UploadFile = File(...)):
    print(f"--- 분석 시작: {file.filename} ---")
    content = await file.read()
    try:
        y_drums, y_bass, y_piano_raw, sr = run_demucs_separation(content)
    except Exception as e:
        return {"error": str(e)}
    
    tempo, _ = librosa.beat.beat_track(y=y_drums, sr=sr)
    actual_tempo = float(tempo[0]) if isinstance(tempo, np.ndarray) else float(tempo)
    if actual_tempo <= 0: actual_tempo = 120.0

    tracks = {"drums": [], "bass": [], "piano": []}

    # 드럼
    onset_env_drums = librosa.onset.onset_strength(y=y_drums, sr=sr)
    peaks_drums = librosa.onset.onset_detect(onset_envelope=onset_env_drums, sr=sr, delta=0.2, wait=10)
    for frame in peaks_drums:
        time_val = librosa.frames_to_time(frame, sr=sr)
        beat_val = float(time_val * (actual_tempo / 60))
        stft = np.abs(librosa.stft(y_drums[frame*512 : frame*512 + 2048]))
        lane = 0 if np.argmax(stft) < 50 else 1
        tracks["drums"].append({"beatNumber": round(beat_val, 2), "lane": lane})

    # 베이스
    chroma_bass = librosa.feature.chroma_stft(y=y_bass, sr=sr)
    onset_env_bass = librosa.onset.onset_strength(y=y_bass, sr=sr)
    peaks_bass = librosa.onset.onset_detect(onset_envelope=onset_env_bass, sr=sr, delta=0.15, wait=12)
    for frame in peaks_bass:
        time_val = librosa.frames_to_time(frame, sr=sr)
        beat_val = float(time_val * (actual_tempo / 60))
        pitch_idx = np.argmax(chroma_bass[:, frame])
        lane = int(pitch_idx % 4)
        tracks["bass"].append({"beatNumber": round(beat_val, 2), "lane": lane})

    # 피아노
    try:
        y_p, _ = librosa.load(SAMPLE_PIANO, sr=sr)
        fp_piano = get_normalized_fp(y_p, sr)
        chroma_piano = librosa.feature.chroma_stft(y=y_piano_raw, sr=sr)
        onset_env_piano = librosa.onset.onset_strength(y=y_piano_raw, sr=sr)
        peaks_piano = librosa.onset.onset_detect(onset_envelope=onset_env_piano, sr=sr, delta=0.14, wait=15)
        mfcc_full = librosa.feature.mfcc(y=y_piano_raw, sr=sr, n_mfcc=40)
        for frame in peaks_piano:
            current_fp = mfcc_full[:, frame]
            current_fp_norm = (current_fp - np.mean(current_fp)) / (np.std(current_fp) + 1e-9)
            sim_p = np.dot(fp_piano, current_fp_norm) / (np.linalg.norm(fp_piano) * np.linalg.norm(current_fp_norm) + 1e-9)
            if sim_p > 0.82:
                time_val = librosa.frames_to_time(frame, sr=sr)
                beat_val = float(time_val * (actual_tempo / 60))
                pitch_idx = np.argmax(chroma_piano[:, frame])
                lane = int(pitch_idx // 3)
                tracks["piano"].append({"beatNumber": round(beat_val, 2), "lane": lane})
    except: pass

    for instrument in tracks:
        tracks[instrument].sort(key=lambda x: x["beatNumber"])
    
    return {"beatsPerMinute": round(actual_tempo, 2), "tracks": tracks}

# [2] UDP 멀티플레이 중계 서버 로직 (UDP 8080)
# ==========================================

class Room:
    def __init__(self, room_id: str):
        self.room_id = room_id
        self.clients: Dict[str, Tuple[str, int]] = {}

class UDPRelayProtocol(asyncio.DatagramProtocol):
    def __init__(self):
        self.rooms: Dict[str, Room] = {}
        self.transport = None

    def connection_made(self, transport):
        self.transport = transport
        print("-> [UDP] 서버 가동 (Port 8080)")

    def datagram_received(self, data, addr):
        try:
            recv_time = time.time()
            packet = json.loads(data.decode('utf-8'))
            p_type = packet.get("type")
            player_id = packet.get("playerId")
            room_id = packet.get("roomId", "Lobby")

            # 방 관리
            if room_id not in self.rooms:
                self.rooms[room_id] = Room(room_id)
            self.rooms[room_id].clients[player_id] = addr

            # 핑 패킷 (동기화)
            if p_type == 2:
                packet["relayTimestamp"] = time.time()
                # T2(수신 시각)를 추가로 보내면 클라이언트에서 처리 지연을 계산 가능
                packet["startTime"] = recv_time 
                self.transport.sendto(json.dumps(packet).encode('utf-8'), addr)

            # 게임 시작 (신뢰성 전송)
            elif p_type == 5:
                print(f"-> [{room_id}] Game Start by {player_id}")
                asyncio.create_task(self.reliable_broadcast(room_id, packet))

            # 일반 중계
            else:
                packet["relayTimestamp"] = time.time()
                updated_data = json.dumps(packet).encode('utf-8')
                self.broadcast(room_id, updated_data, exclude_id=player_id)

        except Exception as e:
            print(f"UDP Error: {e}")

    def broadcast(self, room_id, data, exclude_id=None):
        if room_id in self.rooms:
            for pid, addr in self.rooms[room_id].clients.items():
                if pid != exclude_id:
                    self.transport.sendto(data, addr)

    async def reliable_broadcast(self, room_id, packet, count=5):
        for _ in range(count):
            packet["relayTimestamp"] = time.time()
            data = json.dumps(packet).encode('utf-8')
            self.broadcast(room_id, data)
            await asyncio.sleep(0.1)

@app.on_event("startup")
async def startup_event():
    loop = asyncio.get_running_loop()
    await loop.create_datagram_endpoint(lambda: UDPRelayProtocol(), local_addr=("0.0.0.0", 8080))

if __name__ == "__main__":

    uvicorn.run(app, host="0.0.0.0", port=8000)
